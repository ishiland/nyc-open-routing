     map?.resize()
   }, [map])

9. In the tablet/desktop return block, wrap the `<aside>` in a collapsible container. The `<aside>` element should get:
   - `onTransitionEnd={handleTransitionEnd}`

     - `overflow: 'hidden'`
     - `flexShrink: 0`
     - `position: 'relative'` (for positioning the toggle button)

10. Add a collapse toggle button inside the `<aside>`, BEFORE the Suspense wrapper:
    ```tsx
    <IconButton
      onClick={() => setIsCollapsed(prev => !prev)}
      aria-label={isCollapsed ? "Expand sidebar" : "Collapse sidebar"}
      sx={{
        position: 'absolute',
        top: 8,
        right: 8,
        zIndex: 1,
        minWidth: 44,
        minHeight: 44,
      }}
    >
      {isCollapsed ? <ChevronRight /> : <ChevronLeft />}
    </IconButton>
    ```

11. Wrap the sidebar content in a div that maintains full width to prevent content reflow during transition:
    ```tsx
    <div style={{
      width: isTabletOrBelow ? SIDEBAR_WIDTH_TABLET_PX : SIDEBAR_WIDTH_PX,
      minWidth: isTabletOrBelow ? SIDEBAR_WIDTH_TABLET_PX : SIDEBAR_WIDTH_PX,
    }}>
      <Suspense fallback={<LoadingSpinner message="Loading controls..." />}>
        {sidebar}
      </Suspense>
    </div>
    ```
